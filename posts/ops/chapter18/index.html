关系型数据库

1 安装

1.1 通过二进制包安装

安装依赖

yum install libaio numactl

获取通用二进制包

wget https://downloads.mysql.com/archives/get/p/23/file/mysql-8.0.33-linux-glibc2.28-x86_64.tar.gz

创建 MySQL 组

groupadd -g 27 -o -r mysql

创建 MySQL 用户

useradd -M -N -g mysql -o -r -d /usr/local/mysql/data -s /bin/false -c "MySQL Server" -u 27 mysql

解压二进制包

tar -zxvf mysql-8.0.33-linux-glibc2.28-x86_64.tar.gz
mv mysql-8.0.33-linux-glibc2.28-x86_64 /usr/local/mysql

设置 PATH 变量

export PATH=/usr/local/mysql/bin:$PATH

创建安全目录（用于导入导出）

cd /usr/local/mysql
mkdir mysql-files
chown mysql:mysql mysql-files
chmod 750 mysql-files

创建配置文件

cd /etc
touch my.cnf
chown root:root my.cnf
chmod 644 my.cnf

添加配置到 my.cnf 文件

[mysqld]
datadir=/usr/local/mysql/data
socket=/tmp/mysql.sock
port=3306
log-error=/usr/local/mysql/data/localhost.localdomain.err
user=mysql
secure_file_priv=/usr/local/mysql/mysql-files
local_infile=OFF

创建数据目录

cd /usr/local/mysql
mkdir data
chmod 750 data
chown mysql:mysql data

初始化

cd /usr/local/mysql
bin/mysqld --defaults-file=/etc/my.cnf --initialize

初始化日志（包含初始密码）

cat /usr/local/mysql/data/localhost.localdomain.err

创建 service 配置文件

cd /usr/lib/systemd/system
touch mysqld.service
chmod 644 mysqld.service

添加配置到 mysqld.service 文件

[Unit]
Description=MySQL Server
Documentation=man:mysqld(8)
Documentation=http://dev.mysql.com/doc/refman/en/using-systemd.html
After=network.target
After=syslog.target

[Install]
WantedBy=multi-user.target

[Service]
User=mysql
Group=mysql

# Have mysqld write its state to the systemd notify socket
Type=notify

# Disable service start and stop timeout logic of systemd for mysqld service.
TimeoutSec=0

# Start main service
ExecStart=/usr/local/mysql/bin/mysqld --defaults-file=/etc/my.cnf $MYSQLD_OPTS

# Use this to switch malloc implementation
EnvironmentFile=-/etc/sysconfig/mysql

# Sets open_files_limit
LimitNOFILE = 10000

Restart=on-failure

RestartPreventExitStatus=1

# Set environment variable MYSQLD_PARENT_PID. This is required for restart.
Environment=MYSQLD_PARENT_PID=1

PrivateTmp=false

1.2 通过 Docker 安装

拉取镜像

docker pull mysql:8.0.31

创建容器并启动

docker run -d \
    --name mysql-server \
    -p 3306:3306 \
    -v ./mysql:/var/lib/mysql \
    -e MYSQL_ROOT_PASSWORD=password \
    -e MYSQL_DATABASE=<db> \
    -e MYSQL_USER=<user> \
    -e MYSQL_PASSWORD=<pwd> \
    mysql:8.0.31

2 使用

2.1 基础操作

登录

cd /usr/local/mysql
bin/mysql -uroot -p
# Enter password: (enter the random root password here)

设置密码

alter user 'root'@'localhost' identified by '<password>';
# 带 caching_sha2_password 加密
alter user 'root'@'localhost' identified with 'caching_sha2_password' by '<password>';
